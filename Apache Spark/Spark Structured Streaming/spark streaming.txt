# ==============================================
# PySpark Structured Streaming - JSON to JSON/Parquet
# ==============================================

from pyspark.sql import SparkSession
from pyspark.sql.functions import *
from pyspark.sql.types import *
from pyspark.sql.window import Window


# 1. Create Spark Session
spark = (
    SparkSession.builder
    .appName("Spark Structured Streaming")
    .getOrCreate()
)

# Print Spark version
print(f"Spark version: {spark.version}")


# ==============================================
# 2. Read JSON data in batch
# ==============================================

df = spark.read.json(
    "hdfs://localhost:9000/data/json/json.json"
)

# Print schema
df.printSchema()


# ==============================================
# 3. Get schema to use for streaming
# ==============================================

schema = df.schema


# ==============================================
# 4. Read JSON data as a stream
# ==============================================

df_stream = (
    spark.readStream
    .format("json")
    .schema(schema)
    .option("multiline", True)
    .load("hdfs://localhost:9000/data/json/")
)


# ==============================================
# 5. Transformations
# ==============================================

# Convert timestamp from string to timestamp
df_stream = df_stream.withColumn(
    "timestamp",
    to_timestamp(col("timestamp"))
)

# Calculate total sales for each order
df_stream = df_stream.withColumn(
    "total_sales",
    col("price") * col("quantity")
)

# Filter expensive orders
df_stream = df_stream.filter(
    col("total_sales") >= 500
)

# Select required columns
df_stream = df_stream.select(
    "order_id",
    "customer_id",
    "customer_name",
    "category",
    "product",
    "quantity",
    "price",
    "total_sales",
    "city",
    "timestamp"
)


# ==============================================
# 6. Paths
# ==============================================

json_output_path = "hdfs://localhost:9000/data/json/output"
json_checkpoint = "hdfs://localhost:9000/data/json/checkpoint"

parquet_output_path = "hdfs://localhost:9000/data/sink"
parquet_checkpoint = "hdfs://localhost:9000/data/sink/checkpoint"


# ==============================================
# 7. Write streaming output as JSON
# ==============================================

json_query = (
    df_stream.writeStream
    .format("json")
    .outputMode("append")
    .trigger(availableNow=True)
    .option("path", json_output_path)
    .option("checkpointLocation", json_checkpoint)
    .start()
)

# Wait until JSON streaming query finishes
json_query.awaitTermination()


# ==============================================
# 8. Write streaming output as Parquet
# ==============================================

parquet_query = (
    df_stream.writeStream
    .format("parquet")
    .outputMode("append")
    .trigger(availableNow=True)
    .option("path", parquet_output_path)
    .option("checkpointLocation", parquet_checkpoint)
    .start()
)

# Wait until Parquet streaming query finishes
parquet_query.awaitTermination()


--------------------------------
-- save the output as parquet -- 
--------------------------------
query = (
    df_stream.writeStream
    .format("parquet")
    .outputMode("append")
    .option("path", "hdfs://localhost:9000/data/sink")
    .option("checkpointLocation", "hdfs://localhost:9000/data/sink/checkpoint")
    .start()
)

query.awaitTermination()
